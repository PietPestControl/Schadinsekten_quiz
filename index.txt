<script>
  // Datenbank der Insekten
  const insects = [
    {image: "reismehlkaefer.jpg", name: "Reismehlkäfer"},
    {image: "brotkaefer.jpg", name: "Brotkäfer"},
    {image: "mehlkaefer.jpg", name: "Mehlkäfer"},
    {image: "getreideplattkaefer.jpg", name: "Getreideplattkäfer"},
    {image: "kornkaefer.jpg", name: "Kornkäfer"},
    {image: "reiskaefer.jpg", name: "Reiskäfer"},
    {image: "doerrobstmotte.jpg", name: "Dörrobstmotte"},
    {image: "mehlmotte.jpg", name: "Mehlmotte"},
    {image: "kakaomotte.jpg", name: "Kakaomotte"},
    {image: "mehlmilbe.jpg", name: "Mehlmilbe"},
    {image: "stubenfliege.jpg", name: "Stubenfliege"},
    {image: "schmeissfliege.jpg", name: "Schmeißfliege"},
    {image: "fruchtfliege.jpg", name: "Fruchtfliege"},
    {image: "deutsche_schabe.jpg", name: "Deutsche Schabe"},
    {image: "orientalische_schabe.jpg", name: "Orientalische Schabe"},
    {image: "amerikanische_schabe.jpg", name: "Amerikanische Schabe"},
    {image: "schwarze_wegameise.jpg", name: "Schwarze Wegameise"},
    {image: "pharaoameise.jpg", name: "Pharaoameise"},
    {image: "menschenfloh.jpg", name: "Menschenfloh"},
    {image: "katzenfloh.jpg", name: "Katzenfloh"},
    {image: "hausbock.jpg", name: "Hausbock"},
    {image: "gemeiner_nagekaefer.jpg", name: "Gemeiner Nagekäfer"},
    {image: "kleidermotte.jpg", name: "Kleidermotte"},
    {image: "pelzmotte.jpg", name: "Pelzmotte"},
    {image: "deutsche_wespe.jpg", name: "Deutsche Wespe"},
    {image: "gemeine_wespe.jpg", name: "Gemeine Wespe"},
    {image: "hausmuecke.jpg", name: "Hausmücke"},
    {image: "silberfischchen.jpg", name: "Silberfischchen"},
    {image: "maeuseassel.jpg", name: "Mäuseassel"}
  ];

  const ROUND = 20;
  const $ = id => document.getElementById(id);
  const img = $('insect-image'), scanner = $('scanner'), choicesDiv = $('choices'),
        msg = $('msg'), nextBtn = $('next');

  let deck = [], idx = 0, score = 0, streak = 0, wrong = [], current = null, locked = false;

  // Fisher-Yates statt sort(random): wirklich gleichmäßig gemischt
  function shuffle(a) {
    a = a.slice();
    for (let i = a.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [a[i], a[j]] = [a[j], a[i]];
    }
    return a;
  }

  function loadBest() { try { return localStorage.getItem('insekten-best'); } catch (e) { return null; } }
  function saveBest(v) { try { localStorage.setItem('insekten-best', v); } catch (e) {} }

  function newRound(pool) {
    deck = shuffle(pool || insects).slice(0, ROUND);
    idx = 0; score = 0; streak = 0; wrong = [];
    $('result').classList.add('hidden');
    $('quiz').classList.remove('hidden');
    updateStats();
    showQuestion();
  }

  function updateStats() {
    $('score').textContent = score;
    $('streak').textContent = streak;
    const b = loadBest();
    $('best').textContent = b ? b + '/' + ROUND : '–';
    $('progress').style.width = (idx / deck.length * 100) + '%';
  }

  function showQuestion() {
    locked = false;
    current = deck[idx];
    msg.textContent = ''; msg.className = '';
    nextBtn.classList.add('hidden');

    $('fallback').style.display = 'none';
    img.style.display = '';
    img.onerror = () => { img.style.display = 'none'; $('fallback').style.display = 'grid'; };
    img.src = current.image;
    scanner.classList.remove('scan'); void scanner.offsetWidth; scanner.classList.add('scan');

    const others = shuffle(insects.filter(i => i.name !== current.name)).slice(0, 3);
    const options = shuffle([current, ...others]);

    choicesDiv.innerHTML = '';
    options.forEach((o, n) => {
      const b = document.createElement('button');
      b.className = 'choice';
      b.dataset.name = o.name;
      b.innerHTML = '<kbd>' + (n + 1) + '</kbd><span></span>';
      b.lastChild.textContent = o.name;
      b.onclick = () => answer(b);
      choicesDiv.appendChild(b);
    });
  }

  function answer(btn) {
    if (locked) return;
    locked = true;
    const ok = btn.dataset.name === current.name;
    [...choicesDiv.children].forEach(c => {
      c.disabled = true;
      if (c.dataset.name === current.name) c.classList.add('ok');
    });
    if (ok) {
      score++; streak++;
      msg.textContent = streak >= 3 ? 'Richtig! ' + streak + ' in Folge.' : 'Richtig!';
      msg.className = 'ok';
    } else {
      streak = 0; wrong.push(current);
      btn.classList.add('bad');
      msg.textContent = 'Falsch. Das war: ' + current.name;
      msg.className = 'bad';
    }
    idx++;
    updateStats();
    nextBtn.textContent = idx >= deck.length ? 'Auswertung ansehen' : 'Nächstes Insekt';
    nextBtn.classList.remove('hidden');
    nextBtn.focus();
  }

  function rankFor(p) {
    if (p >= .95) return 'Rang: Großmeister der Schädlingsbekämpfung';
    if (p >= .8)  return 'Rang: Prüfungsreif';
    if (p >= .6)  return 'Rang: Auf gutem Weg';
    return 'Rang: Noch etwas üben';
  }

  function finish() {
    $('quiz').classList.add('hidden');
    $('result').classList.remove('hidden');
    $('progress').style.width = '100%';
    const total = deck.length;
    $('r-title').textContent = score + ' von ' + total + ' richtig';
    $('r-rank').textContent = rankFor(score / total);
    const b = parseInt(loadBest() || '0', 10);
    if (total === ROUND && score > b) { saveBest(score); updateStats(); }

    const w = $('r-wrong');
    if (wrong.length) {
      w.innerHTML = '<p>Das solltest du noch einmal ansehen:</p><ul>' +
        wrong.map(i => '<li><b>' + i.name + '</b></li>').join('') + '</ul>';
    } else {
      w.innerHTML = '<p>Keine Fehler. Sauber gescannt.</p>';
    }
    $('redo').classList.toggle('hidden', wrong.length === 0);
  }

  nextBtn.onclick = () => (idx >= deck.length ? finish() : showQuestion());
  $('again').onclick = () => newRound();
  $('redo').onclick = () => newRound(wrong);

  // Tastatur: 1–4 wählt, Enter geht weiter
  document.addEventListener('keydown', e => {
    if ($('quiz').classList.contains('hidden')) return;
    const n = parseInt(e.key, 10);
    if (n >= 1 && n <= 4 && !locked && choicesDiv.children[n - 1]) answer(choicesDiv.children[n - 1]);
  });

  newRound();
</script>
